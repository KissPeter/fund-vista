controls are disabled let pending = false; // slider moved mid-flight → one follow-up convert after let depth = 0; // convert() recursion depth (the §4.3 retry nests once) const $ = (id) => document.getElementById(id); const statusEl = $("status"); const setStatus = (msg, cls) => { statusEl.textContent = msg; statusEl.className = cls || ""; }; function readParams() { const num = (id) => parseFloat($(id).value); return { methods: ["contour", "hatch", "flow"].filter((m) => $("m_" + m).checked), threshold: parseInt($("threshold").value, 10), blur_radius: num("blur_radius"), contrast: num("contrast"), brightness: num("brightness"), remove_background: $("remove_background").checked, strip_hatch_px: parseInt($("strip_hatch_px").value, 10), hatch_pitch_mm: num("hatch_pitch_mm"), hatch_angle_deg: num("hatch_angle_deg"), contour_simplify: num("contour_simplify"), linemerge_tolerance_mm: num("linemerge_tolerance_mm"), linesimplify_tolerance_mm: num("linesimplify_tolerance_mm"), linesort: $("linesort").checked, reloop_tolerance_mm: num("reloop_tolerance_mm"), page: { size: $("page_size").value, orientation: $("page_orientation").value, margin_mm: num("page_margin_mm"), frame: $("page_frame").checked, frame_radius_mm: num("page_frame_radius_mm"), }, pen: { draw_speed_mm_s: num("draw_speed_mm_s"), travel_speed_mm_s: num("travel_speed_mm_s"), pen_lift_s: num("pen_lift_s"), }, label: { enabled: $("label_enabled").checked, text: $("label_text").value, align: $("label_align").value, height_mm: num("label_height_mm"), font: $("label_font").value, border: $("label_border").checked, pad_left_mm: num("label_pad_left_mm"), pad_right_mm: num("label_pad_right_mm"), border_radius_mm: num("label_border_radius_mm"), }, line_color: $("line_color").value, background: $("background").value, }; } async function upload(file) { const form = new FormData(); form.append("file", file); const resp = await fetch("/v1/images", { method: "POST", body: form }); const body = await resp.json(); if (!resp.ok) throw new Error(body.error ? body.error.code + ": " + body.error.message : "upload failed"); imageId = body.image_id; $("meta").textContent = body.image_id.slice(0, 12) + "… " + body.format + " " + body.width + "×" + body.height + (body.is_vector ? " (vector)" : "") + (body.warnings.length ? " ⚠ " + body.warnings.join(", ") : ""); return body; } async function convert(retried) { if (!imageId) { setStatus("Upload an image first."); return; } depth += 1; setBusy(true); try { setStatus("Converting…"); const resp = await fetch("/v1/convert", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ image_id: imageId, params: readParams() }), }); const body = await resp.json(); if (resp.ok) { render(body); return; } const code = body.error ? body.error.code : ""; // Spec §4.3: expired cache → silent re-upload with the retained file, retry once. if (resp.status === 404 && code === "image_not_found" && currentFile && !retried) { setStatus("Cache expired, re-uploading…"); await upload(currentFile); return convert(true); } if (resp.status === 429) { setStatus("Rate limited — retry in a few seconds.", "error"); return; } setStatus("Convert failed (" + code + "): " + (body.error ? body.error.message : ""), "error"); } finally { depth -= 1; if (depth === 0) drain(); // only the outermost call re-enables + follows up } } function setBusy(on) { busy = on; document.querySelectorAll("#params input, #params select, #params button, #file") .forEach((el) => { el.disabled = on; }); $("params").classList.toggle("busy", on); } function drain() { // End of an in-flight op: re-enable, then re-arm the SAME debounce timer // (not an immediate convert) if a slider moved mid-flight. The user may // still be mid-drag exactly when the prior request resolves, so firing // right away would convert on a half-adjusted value; scheduleConvert() // gives them a fresh debounce window instead. setBusy(false); if (pending && imageId) { pending = false; scheduleConvert(); } else { pending = false; } } function render(body) { setStatus("Done."); $("preview").src = body.svg_url; $("download").href = body.svg_url; const s = body.stats; $("st_strokes").textContent = s.strokes; $("st_points").textContent = s.points.before + " → " + s.points.after; $("st_segments").textContent = s.segments.before + " → " + s.segments.after; $("st_down").textContent = s.pen_down_mm; $("st_up").textContent = s.pen_up_mm; $("st_time").textContent = s.estimated_time_s; $("stats").hidden = false; $("vpype").textContent = body.vpype_command; const ul = $("warnings"); ul.innerHTML = ""; for (const w of body.warnings || []) { const li = document.createElement("li"); li.textContent = w; li.className = "warn"; ul.appendChild(li); } } function scheduleConvert() { clearTimeout(timer); timer = setTimeout(() => { // A queued timer firing mid-flight must not overlap: coalesce into one // follow-up drained when the in-flight op finishes. if (busy) { pending = true; return; } convert(false).catch((e) => setStatus(String(e), "error")); }, 500); // spec §4.2: "~300-500ms debounce" — upper end of the stated range } $("params").addEventListener("input", scheduleConvert); // spec §4.2: debounce, convert only function syncOutputs() { document.querySelectorAll('#params input[type="range"]').forEach((el) => { const out = $(el.id + "_val"); if (out) out.textContent = el.value; }); } $("params").addEventListener("input", syncOutputs); syncOutputs(); $("resetBtn").addEventListener("click", () => { $("params").reset(); // restores server-rendered defaults, checkboxes included syncOutputs(); scheduleConvert(); });
